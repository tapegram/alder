# users and logging in
alder new table User email:Text.Email@unique name:Text.NonEmpty passwordHash:Text created:Time@created --no-timestamps
alder new auth --users User

# notes, owned by their writer
alder new table Note title:Text.NonEmpty body:Text owner:User.UserId@ref=User@owner created:Time@created --no-timestamps

# one note's page
alder new page Notes.Show '/notes/{id}' --param id:Note.NoteId --requires user --blank
alder fill --decl Notes.Show.Model - <<'EOF'
{ note: Note }
EOF
alder fill Notes.Show.load - <<'EOF'
{
  let note <- Db.getMine Note params.id
  Query.succeed { note: note }
}
EOF
alder fill Notes.Show.view - <<'EOF'
<Layout.Page title={model.note.title}>
  <h1 id="title">{model.note.title}</h1>
  <p id="body">{model.note.body}</p>
  <a href={Notes.Index {}}>Back</a>
</Layout.Page>
EOF

# the list of my notes
alder new page Notes.Index / --requires user --blank
alder fill --decl Notes.Index.Model - <<'EOF'
{ notes: List Note }
EOF
alder fill Notes.Index.load - <<'EOF'
{
  let notes <- Db.mine Note |> Db.orderBy Note.created.desc |> Db.all
  Query.succeed { notes: notes }
}
EOF
alder fill Notes.Index.view - <<'EOF'
<h1>Notes</h1>
EOF
alder new form Notes.Index NewNote title:Text.NonEmpty body:Text --msg Add
alder new msg Notes.Index Delete id:Note.NoteId
alder fill Notes.Index.update.Add - <<'EOF'
{
  let _ <- Db.insertMine Note { title: p.form.title, body: p.form.body }
  Task.succeed (Redirect (Notes.Index {}))
}
EOF
alder fill Notes.Index.update.Delete - <<'EOF'
{
  let _ <- Db.deleteMine Note p.id
  Task.succeed (Redirect (Notes.Index {}))
}
EOF
alder fill --decl Notes.Index.view - <<'EOF'
<Layout.Page title="Notes">
  <h1>Notes</h1>
  <form submit={Add}>
    <Errors form />
    <Input field="title" label="Title" />
    <Textarea field="body" label="Body" />
    <button>Add</button>
  </form>
  <ul id="notes">
    {for note in model.notes ->
      <li>
        <a class="title" href={Notes.Show { id: note.id }}>{note.title}</a>
        <button click={Delete { id: note.id }}>Delete</button>
      </li>}
  </ul>
  <button click={App.Logout}>Log out</button>
</Layout.Page>
EOF

# after login, go to the list
alder fill Auth.Login.update - --hole ?afterLogin <<'EOF'
Notes.Index {}
EOF

# tests of deleting
alder new test Notes.Index "deleting a note of another user is not found"
alder fill 'Notes.Index."deleting a note of another user is not found"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let other <- Db.insert User { email: Text.Email "other@example.com", name: Text.NonEmpty "Other", passwordHash: "" }
  let id <- Db.insert Note { title: Text.NonEmpty "Mine", body: "secret", owner: me }
  let _ <- Test.login { user: Just other }
  let r <- Test.run (update (Delete { id: id }) { notes: [] } { user: other })
  let after <- Db.get Note id
  Task.succeed ({ value: r.value, note: Maybe.map (\n -> n.title) after } == { value: Err "not found", note: Just (Text.NonEmpty "Mine") })
}
EOF
alder new test Notes.Index "deleting a missing note is not found"
alder fill 'Notes.Index."deleting a missing note is not found"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let _ <- Test.login { user: Just me }
  let r <- Test.run (update (Delete { id: Note.NoteId "missing" }) { notes: [] } { user: me })

}
EOF
alder new test Notes.Index "the owner can delete a note"
alder fill 'Notes.Index."the owner can delete a note"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let id <- Db.insert Note { title: Text.NonEmpty "Mine", body: "secret", owner: me }
  let _ <- Test.login { user: Just me }
  let _ <- Test.get (Notes.Index {})
  let r <- Test.click (Delete { id: id })
  let after <- Db.get Note id
  Task.succeed ({ answer: r.answer, note: Maybe.map (\n -> n.title) after } == { answer: Redirected (Notes.Index {}), note: Nothing })
}
EOF
alder new test Notes.Index "adding a note lists only my notes"
alder fill 'Notes.Index."adding a note lists only my notes"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let other <- Db.insert User { email: Text.Email "other@example.com", name: Text.NonEmpty "Other", passwordHash: "" }
  let _ <- Db.insert Note { title: Text.NonEmpty "Theirs", body: "", owner: other }
  let _ <- Test.login { user: Just me }
  let _ <- Test.get (Notes.Index {})
  let added <- Test.submit NewNote { title: Text.NonEmpty "Mine", body: "hello" }
  let page <- Test.get (Notes.Index {})
  Task.succeed ({ answer: added.answer, titles: Test.texts "#notes a.title" page.doc } == { answer: Redirected (Notes.Index {}), titles: ["Mine"] })
}
EOF
