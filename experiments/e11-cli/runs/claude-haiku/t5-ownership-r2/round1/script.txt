# the note page: one note, shown only to its owner
alder new page Notes.Show "/notes/{id}" --show Note --requires user
alder fill --decl Notes.Show.Model - <<'EOF'
{ note: Note }
EOF
alder fill --decl Notes.Show.load - <<'EOF'
{
  let found <- Db.get Note params.id
  match found {
    Just n -> if n.owner == session.user then Query.succeed { note: n } else Query.notFound,
    Nothing -> Query.notFound,
  }
}
EOF
alder fill --decl Notes.Show.view - <<'EOF'
<Layout.Page title={model.note.title}>
  <h1 id="title">{model.note.title}</h1>
  <p id="body">{model.note.body}</p>
  <a href={Notes.Index {}}>Back to notes</a>
</Layout.Page>
EOF

# the delete helper: only the owner's note is deleted, anything else is not found
alder new decl Notes.Index deleteOwn "User.UserId -> Note.NoteId -> Task Unit"
alder fill Notes.Index.deleteOwn - <<'EOF'
\me id -> {
  let found <- Db.get Note id
  match found {
    Just n -> if n.owner == me then Db.delete Note id else Task.notFound,
    Nothing -> Task.notFound,
  }
}
EOF

# the page's update and view
alder fill --decl Notes.Index.update - <<'EOF'
match msg {
  Add p -> {
    let _ <- Db.insert Note { title: p.form.title, body: p.form.body, owner: session.user }
    Task.succeed (Redirect (Notes.Index {}))
  },
  Delete p -> {
    let _ <- deleteOwn session.user p.id
    Task.succeed (Redirect (Notes.Index {}))
  },
}
EOF
alder fill --decl Notes.Index.view - <<'EOF'
<Layout.Page title="Notes">
  <h1>Notes</h1>

  <al-card>
    <form submit={Add}>
      <al-input field="title" label="Title" />
      <al-textarea field="body" label="Body" />
      <al-alert variant="destructive"><Errors form /></al-alert>
      <al-button type="submit">Add</al-button>
    </form>
  </al-card>

  <ul id="notes">
    {for note in model.notes ->
      <li>
        <a class="title" href={Notes.Show { id: note.id }}>{note.title}</a>
        <al-button variant="ghost" size="sm" click={Delete { id: note.id }}>Delete</al-button>
      </li>}
  </ul>

  <al-button variant="outline" click={App.Logout}>Log out</al-button>
</Layout.Page>
EOF
