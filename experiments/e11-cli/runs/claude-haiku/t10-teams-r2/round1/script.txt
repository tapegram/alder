# Teams.Show: the page, its messages, and its form
alder fill Teams.Show.view - --hole ?view <<'EOF'
<p>placeholder</p>
EOF
alder new msg Teams.Show Remove user:User.UserId
alder new form Teams.Show NewInvite email:Text.Email --msg SendInvite
alder fill --decl Teams.Show.Model - <<'EOF'
{ team: Team, people: List User, viewerIsOwner: Bool }
EOF
alder fill Teams.Show.load - --hole ?load <<'EOF'
{
  let found <- Db.get Team params.id
  let members <- Db.from Member |> Db.where (Member.team.eq params.id) |> Db.all
  match found {
    Nothing -> Query.notFound,
    Just team -> {
      let owned = team.owner == session.user
      let inTeam = owned || List.any (\m -> m.user == session.user) members
      if inTeam then {
        let ownerRow <- Db.get User team.owner
        let memberRows <- Query.all (List.map (\m -> Db.get User m.user) members)
        let others = List.sortByText (\u -> u.name) (List.filterMap (\u -> u) memberRows)
        Query.succeed { team: team, people: Maybe.toList ownerRow ++ others, viewerIsOwner: owned }
      } else Query.notFound
    },
  }
}
EOF
alder fill --decl Teams.Show.update - <<'EOF'
match msg {
  Remove p -> {
    let found <- Db.from Member |> Db.where (Member.team.eq model.team.id) |> Db.where (Member.user.eq p.user) |> Db.first
    match found {
      Nothing -> Task.notFound,
      Just m -> {
        if session.user == model.team.owner && p.user != session.user then {
          let _ <- Db.delete Member m.id
          Task.succeed (Redirect (Teams.Show { id: model.team.id }))
        } else Task.notFound
      },
    }
  },
  SendInvite p -> {
    if session.user == model.team.owner then {
      let waiting <- Db.from Invite |> Db.where (Invite.team.eq model.team.id) |> Db.where (Invite.email.eq p.form.email) |> Db.where (Invite.status.eq Pending) |> Db.count
      if waiting == 0 then {
        let _ <- Db.insert Invite { team: model.team.id, email: p.form.email }
        Task.succeed (Redirect (Teams.Show { id: model.team.id }))
      } else Task.succeed (Redirect (Teams.Show { id: model.team.id }))
    } else Task.notFound
  },
}
EOF

# Invites.Index: the waiting invites and their answers
alder new msg Invites.Index Accept id:Invite.InviteId
alder new msg Invites.Index Decline id:Invite.InviteId
alder fill --decl Invites.Index.Model - <<'EOF'
{ invites: List Invite, teams: List Team }
EOF
alder fill --decl Invites.Index.load - <<'EOF'
{
  let me <- Db.from User |> Db.where (User.id.eq session.user) |> Db.first
  let email = Maybe.map (\u -> u.email) me
  let pending <- Db.from Invite |> Db.where (Invite.status.eq Pending) |> Db.all
  let invites = List.filter (\i -> email == Just i.email) pending
  let teams <- Db.from Team |> Db.all
  Query.succeed { invites: invites, teams: List.filter (\t -> List.any (\i -> i.team == t.id) invites) teams }
}
EOF
alder fill --decl Invites.Index.update - <<'EOF'
match msg {
  Accept p -> {
    let found <- Db.get Invite p.id
    let me <- Db.from User |> Db.where (User.id.eq session.user) |> Db.first
    match found {
      Nothing -> Task.notFound,
      Just i -> {
        if i.status == Pending && Maybe.map (\u -> u.email) me == Just i.email then {
          let _ <- Db.update Invite p.id (\x -> { x | status: Accepted })
          let team <- Db.get Team i.team
          let owner = Maybe.map (\t -> t.owner) team == Just session.user
          let joined <- Db.from Member |> Db.where (Member.team.eq i.team) |> Db.where (Member.user.eq session.user) |> Db.count
          if owner || joined > 0 then Task.succeed (Redirect (Teams.Show { id: i.team })) else {
            let _ <- Db.insert Member { team: i.team, user: session.user }
            Task.succeed (Redirect (Teams.Show { id: i.team }))
          }
        } else Task.notFound
      },
    }
  },
  Decline p -> {
    let found <- Db.get Invite p.id
    let me <- Db.from User |> Db.where (User.id.eq session.user) |> Db.first
    match found {
      Nothing -> Task.notFound,
      Just i -> {
        if i.status == Pending && Maybe.map (\u -> u.email) me == Just i.email then {
          let _ <- Db.update Invite p.id (\x -> { x | status: Declined })
          Task.succeed (Redirect (Invites.Index {}))
        } else Task.notFound
      },
    }
  },
}
EOF
alder fill --decl Invites.Index.view - <<'EOF'
