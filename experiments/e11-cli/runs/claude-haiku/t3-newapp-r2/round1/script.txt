# the login (App's session gets the user; the auth pages are created)
alder new auth --users User

# the front page: published posts, newest first
alder remove page Home
alder new page Home / --list Post
alder fill --decl Home.Model - <<'EOF'
{ posts: List Post }
EOF
alder fill --decl Home.load - <<'EOF'
\params session -> {
  let posts <-
    Db.from Post
      |> Db.where (Post.published.eq True)
      |> Db.orderBy Post.created.desc
      |> Db.all
  Query.succeed { posts: posts }
}
EOF
alder fill --decl Home.view - <<'EOF'
\model session ->
  <Layout.Page title="Blog">
    <h1>Blog</h1>
    <ul>
      {for p in model.posts -> <li><a href={Posts.Show { slug: p.slug }}>{p.title}</a></li>}
    </ul>
    {if session.user == Nothing then <a href={Auth.Login {}}>Log in</a> else Html.none}
  </Layout.Page>
EOF
alder new test Home "the front page lists published posts newest first"
alder fill 'Home."the front page lists published posts newest first"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let _ <- Db.insert Post { slug: Post.Slug "first", title: Text.NonEmpty "First", body: "a", published: True, author: me }
  let _ <- Db.insert Post { slug: Post.Slug "draft", title: Text.NonEmpty "Draft", body: "b", published: False, author: me }
  let _ <- Db.insert Post { slug: Post.Slug "second", title: Text.NonEmpty "Second", body: "c", published: True, author: me }
  let page <- Test.get (Home {})
  Task.succeed (Test.texts "li a" page.doc == ["Second", "First"])
}
EOF

# a post with its comments, and the comment form
alder remove page Posts.Show
alder new page Posts.Show '/posts/{slug}' --show Post --by slug --param slug:Post.Slug
alder new form Posts.Show NewComment author:Text.NonEmpty body:Text.NonEmpty --into Comment --msg AddComment
alder fill --decl Posts.Show.Model - <<'EOF'
{ post: Post, comments: List Comment }
EOF
alder fill --decl Posts.Show.load - <<'EOF'
\params session -> {
  let found <- Db.from Post |> Db.where (Post.slug.eq params.slug) |> Db.first
  match found {
    Nothing -> Query.notFound,
    Just post -> {
      let visible = post.published || session.user == Just post.author
      if visible then {
        let comments <-
          Db.from Comment
            |> Db.where (Comment.post.eq post.id)
            |> Db.orderBy Comment.created.asc
            |> Db.all
        Query.succeed { post: post, comments: comments }
      } else Query.notFound
    },
  }
}
EOF
alder fill --decl Posts.Show.view - <<'EOF'
\model session ->
  <Layout.Page title={model.post.title}>
    <h1>{model.post.title}</h1>
    <p>{model.post.body}</p>
    <h2>Comments</h2>
    <ul id="comments">
      {for c in model.comments -> <li><strong>{c.author}</strong>: {c.body}</li>}
    </ul>
    <al-card>
      <form submit={AddComment}>
        <al-input field="author" label="Your name" />
        <Errors field="author" />
        <al-textarea field="body" label="Comment" />
        <Errors field="body" />
        <Errors form />
        <al-button type="submit">Post comment</al-button>
      </form>
    </al-card>
  </Layout.Page>
EOF
alder fill --decl Posts.Show.update - <<'EOF'
\msg model session -> match msg {
  AddComment p -> {
    let _ <- Db.insert Comment { post: model.post.id, author: p.form.author, body: p.form.body }
    Task.succeed Rerender
  },
}
EOF
alder new test Posts.Show "an unpublished post is not found for a reader"
alder fill 'Posts.Show."an unpublished post is not found for a reader"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let _ <- Db.insert Post { slug: Post.Slug "draft", title: Text.NonEmpty "Draft", body: "b", published: False, author: me }
  let r <- Test.open (Posts.Show { slug: Post.Slug "draft" })
  Task.succeed (r.answer == PageMissing)
}
EOF
alder new test Posts.Show "a comment with an empty author is refused"
alder fill 'Posts.Show."a comment with an empty author is refused"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let _ <- Db.insert Post { slug: Post.Slug "hello", title: Text.NonEmpty "Hello", body: "Hi", published: True, author: me }
  let _ <- Test.get (Posts.Show { slug: Post.Slug "hello" })
  let r <- Test.submitRaw NewComment [{ name: "author", value: "" }, { name: "body", value: "Nice" }]
  let n <- Db.from Comment |> Db.count
  Task.succeed ({ answer: r.answer, comments: n } == { answer: FormRejected [{ field: "author", message: "must not be empty" }], comments: 0 })
}
EOF

# the admin page: login required, all posts, create and publish toggle
alder new page Admin /admin --list Post --requires user
alder new form Admin NewPost title:Text.NonEmpty slug:Post.Slug body:Text --msg CreatePost
alder new msg Admin TogglePublished id:Post.PostId
alder fill --decl Admin.Model - <<'EOF'
{ posts: List Post }
EOF
alder fill --decl Admin.load - <<'EOF'
\params session -> {
  let posts <- Db.from Post |> Db.orderBy Post.created.desc |> Db.all
  Query.succeed { posts: posts }
}
EOF
alder fill --decl Admin.view - <<'EOF'
\model session ->
  <Layout.Page title="Admin">
    <h1>Posts</h1>
    <al-card>
      <form submit={CreatePost}>
        <al-input field="title" label="Title" />
        <Errors field="title" />
        <al-input field="slug" label="Slug" />
        <Errors field="slug" />
        <al-textarea field="body" label="Body" />
        <Errors field="body" />
        <al-button type="submit">Create post</al-button>
      </form>
    </al-card>
    <ul id="posts">
      {for p in model.posts ->
        <li>
          <a href={Posts.Show { slug: p.slug }}>{p.title}</a>
          <al-button variant="secondary" size="sm" click={TogglePublished { id: p.id }}>{if p.published then "Unpublish" else "Publish"}</al-button>
        </li>}
    </ul>
    <al-button variant="outline" click={App.Logout}>Log out</al-button>
  </Layout.Page>
EOF
alder fill --decl Admin.update - <<'EOF'
\msg model session -> match msg {
  CreatePost p -> {
    let _ <- Db.insert Post { slug: p.form.slug, title: p.form.title, body: p.form.body, author: session.user }
    Task.succeed Rerender
  },
  TogglePublished p -> {
    let _ <- Db.update Post p.id (\x -> { x | published: not x.published })
    Task.succeed Rerender
  },
}
EOF
