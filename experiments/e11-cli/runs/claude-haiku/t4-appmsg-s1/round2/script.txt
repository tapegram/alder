# the theme switch: an app-level message, flipping the session's theme
alder new msg App ToggleTheme
alder fill --decl App.update - <<'EOF'
\msg session -> match msg {
  Logout p -> {
    let _ <- Session.put (\s -> { s | user: Nothing })
    Task.succeed (Redirect (Auth.Login {}))
  },
  ToggleTheme p -> {
    let _ <- Session.put (\s -> { s | theme: if s.theme == Light then Dark else Light })
    Task.succeed Rerender
  },
}
EOF

# the shared shell: the body class, the switch button and the log out button on every page
alder new decl Shell Page "{ title: Text, theme: App.Theme, children: List (Html a) } -> Html a"
alder fill --decl Shell.Page - <<'EOF'
\props ->
  <html>
    <head>
      <meta charset="utf-8" />
      <title>{props.title}</title>
      <link rel="stylesheet" href={Url.external "/static/al/al.css"} />
    </head>
    <body class={if props.theme == Dark then Just "dark" else Nothing}>
      <al-page width="narrow">
        <main>{props.children}</main>
        <nav>
          <al-button variant="outline" click={App.ToggleTheme}>{if props.theme == Dark then "Switch to light" else "Switch to dark"}</al-button>
          <al-button variant="outline" click={App.Logout}>Log out</al-button>
        </nav>
      </al-page>
    </body>
  </html>
EOF

# the notes list: the form that creates a note, and the list of the user's own notes
alder new form Notes.Index NewNote title:Text.NonEmpty body:Text
alder fill --decl Notes.Index.Model - <<'EOF'
{ notes: List Note }
EOF
alder fill --decl Notes.Index.load - <<'EOF'
\params session -> {
  let notes <- Db.mine Note |> Db.orderBy Note.created.desc |> Db.all
  Query.succeed { notes: notes }
}
EOF
alder fill --decl Notes.Index.update - <<'EOF'
\msg model session -> match msg {
  SubmitNewNote p -> {
    let _ <- Db.insertMine Note { title: p.form.title, body: p.form.body }
    Task.succeed (Redirect (Notes.Index {}))
  },
}
EOF
alder fill --decl Notes.Index.view - <<'EOF'
\model session ->
  <Shell.Page title="Notes" theme={session.theme}>
    <h1>Notes</h1>
    <al-card>
      <form submit={SubmitNewNote}>
        <Errors form />
        <al-input field="title" label="Title" />
        <al-textarea field="body" label="Body" />
        <al-button type="submit">Add note</al-button>
      </form>
    </al-card>
    <ul id="notes">
      {for note in model.notes -> <li><a href={Notes.Show { id: note.id }}>{note.title}</a></li>}
    </ul>
  </Shell.Page>
EOF

# one note: its page, editing and deleting it, all for the owner only
alder new page Notes.Show '/notes/{id}' --show Note --requires user
alder new form Notes.Show EditNote title:Text.NonEmpty body:Text
alder new msg Notes.Show DeleteNote
alder fill --decl Notes.Show.Model - <<'EOF'
{ note: Note }
EOF
alder fill --decl Notes.Show.load - <<'EOF'
\params session -> {
  let note <- Db.getMine Note params.id
  Query.succeed { note: note }
}
EOF
alder fill --decl Notes.Show.update - <<'EOF'
\msg model session -> match msg {
  SubmitEditNote p -> {
    let _ <- Db.updateMine Note model.note.id (\n -> { n | title: p.form.title, body: p.form.body })
    Task.succeed (Redirect (Notes.Show { id: model.note.id }))
  },
  DeleteNote p -> {
    let _ <- Db.deleteMine Note model.note.id
    Task.succeed (Redirect (Notes.Index {}))
  },
}
EOF
alder fill --decl Notes.Show.view - <<'EOF'
\model session ->
  <Shell.Page title={model.note.title} theme={session.theme}>
    <h1>{model.note.title}</h1>
    <al-card>
      <form submit={SubmitEditNote}>
        <Errors form />
        <al-input field="title" label="Title" value={model.note.title} />
        <al-textarea field="body" label="Body" value={model.note.body} />
        <al-button type="submit">Save</al-button>
      </form>
    </al-card>
    <al-button variant="destructive" click={DeleteNote}>Delete</al-button>
    <a href={Notes.Index {}}>Back to notes</a>
  </Shell.Page>
EOF

# the theme flips: Light to Dark and back, seen in the body's class
alder new test App "the theme switch flips the theme"
alder fill 'App."the theme switch flips the theme"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let _ <- Test.login { user: Just me, theme: Light }
  let _ <- Test.get (Notes.Index {})
  let once <- Test.click ToggleTheme
  let twice <- Test.click ToggleTheme
  Task.succeed ({ dark: Test.attr "class" "body" once.doc, back: Test.attr "class" "body" twice.doc, theme: twice.session.theme } == { dark: Just "dark", back: Nothing, theme: Light })
}
EOF
