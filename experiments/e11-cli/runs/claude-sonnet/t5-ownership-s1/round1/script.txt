# login first: the session needs a user before any page can require one
alder new auth --users User

# the note's page
alder new page Notes.Show '/notes/{id}' --blank --requires user --param id:Note.NoteId
alder fill Notes.Show.Model - <<'EOF'
{ note: Note }
EOF
alder fill Notes.Show.load - <<'EOF'
\params session -> {
  let found <- Db.from Note |> Db.where (Note.id.eq params.id) |> Db.where (Note.owner.eq session.user) |> Db.first
  match found {
    Just n -> Query.succeed { note: n },
    Nothing -> Query.notFound,
  }
}
EOF
alder fill Notes.Show.view - <<'EOF'
\model session ->
  <Layout.Page title={model.note.title}>
    <h1 id="title">{model.note.title}</h1>
    <p id="body">{model.note.body}</p>
    <a href={Notes.Index {}}>Back</a>
  </Layout.Page>
EOF

# the list page
alder new page Notes.Index / --blank --requires user
alder fill --decl Notes.Index.Model - <<'EOF'
{ notes: List Note }
EOF
alder fill Notes.Index.load - <<'EOF'
\params session -> {
  let notes <- Db.from Note |> Db.where (Note.owner.eq session.user) |> Db.orderBy Note.created.desc |> Db.all
  Query.succeed { notes: notes }
}
EOF
alder fill Notes.Index.view - <<'EOF'
\model session -> <p>Notes</p>
EOF
alder new form Notes.Index NewNote title:Text.NonEmpty body:Text --msg Add
alder new msg Notes.Index Delete id:Note.NoteId
alder fill --decl Notes.Index.update - <<'EOF'
\msg model session -> match msg {
  Add p -> {
    let _ <- Db.insert Note { title: p.form.title, body: p.form.body, owner: session.user }
    Task.succeed (Redirect (Notes.Index {}))
  },
  Delete p -> {
    let found <- Db.from Note |> Db.where (Note.id.eq p.id) |> Db.where (Note.owner.eq session.user) |> Db.first
    match found {
      Just n -> {
        let _ <- Db.delete Note n.id
        Task.succeed (Redirect (Notes.Index {}))
      },
      Nothing -> Task.notFound,
    }
  },
}
EOF
alder fill --decl Notes.Index.view - <<'EOF'
\model session ->
  <Layout.Page title="Notes">
    <h1>Notes</h1>
    <form submit={Add}>
      <Errors form />
      <Input field="title" label="Title" />
      <Textarea field="body" label="Body" />
      <button>Add</button>
    </form>
    <ul id="notes">
      {for n in model.notes ->
        <li>
          <a class="title" href={Notes.Show { id: n.id }}>{n.title}</a>
          <button click={Delete { id: n.id }}>Delete</button>
        </li>}
    </ul>
  </Layout.Page>
EOF

# tests
alder new test Notes.Index "deleting another user's note is not found"
alder fill 'Notes.Index."deleting another user'"'"'s note is not found"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let other <- Db.insert User { email: Text.Email "other@example.com", name: Text.NonEmpty "Other", passwordHash: "" }
  let id <- Db.insert Note { title: Text.NonEmpty "T", body: "b", owner: me }
  let r <- Test.run (update (Delete { id: id }) { notes: [] } { user: other })
  let after <- Db.get Note id
  Task.succeed ({ value: r.value, there: Maybe.isJust after } == { value: Err "not found", there: True })
}
EOF
