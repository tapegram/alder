# users and login
alder new table User email:Text.Email@unique name:Text.NonEmpty passwordHash:Text created:Time@created --no-timestamps
alder new auth --users User

# rooms and bookings
alder new table Room name:Text.NonEmpty --no-timestamps
alder new table Booking room:Room.RoomId@ref=Room owner:User.UserId@ref=User from:Date to:Date --no-timestamps

# helpers on Booking
alder new decl Booking nextFree "Date -> List Booking -> Date"
alder fill Booking.nextFree - <<'EOF'
\today bs -> List.foldl (\b d -> if b.from <= d && d < b.to then b.to else d) today (List.sortByText (\b -> Date.toText b.from) bs)
EOF
alder new decl Booking conflicts "Date -> Date -> List Booking -> Bool"
alder fill Booking.conflicts - <<'EOF'
\f t bs -> List.any (\b -> f < b.to && b.from < t) bs
EOF

# Rooms.Index
alder new page Rooms.Index /rooms --blank --requires user
alder fill --decl Rooms.Index.Model - <<'EOF'
{ rooms: List { id: Room.RoomId, name: Text, free: Date } }
EOF
alder fill Rooms.Index.load - <<'EOF'
{
  let rooms <- Db.from Room |> Db.orderBy Room.name.asc |> Db.all
  let bookings <- Db.from Booking |> Db.all
  let today <- Date.today
  Query.succeed { rooms: List.map (\r -> { id: r.id, name: r.name, free: Booking.nextFree today (List.filter (\b -> b.room == r.id) bookings) }) rooms }
}
EOF
alder fill Rooms.Index.view - <<'EOF'
<Layout.Page title="Rooms">
  <h1>Rooms</h1>
  <ul id="rooms">
    {for r in model.rooms ->
      <li><a class="name" href={Rooms.Show { id: r.id }}>{r.name}</a> <span class="free">{Date.toText r.free}</span></li>}
  </ul>
</Layout.Page>
EOF

# Rooms.Show
alder new page Rooms.Show '/rooms/{id}' --blank --requires user --param id:Room.RoomId
alder fill --decl Rooms.Show.Model - <<'EOF'
{ room: Room, bookings: List Booking }
EOF
alder fill Rooms.Show.load - <<'EOF'
{
  let found <- Db.get Room params.id
  match found {
    Nothing -> Query.notFound,
    Just r -> {
      let bs <- Db.from Booking |> Db.where (Booking.room.eq r.id) |> Db.orderBy Booking.from.asc |> Db.all
      Query.succeed { room: r, bookings: bs }
    },
  }
}
EOF
alder fill Rooms.Show.view - <<'EOF'
<Layout.Page title="Room">
  <h1>Room</h1>
</Layout.Page>
EOF
alder new form Rooms.Show NewBooking from:Date to:Date --msg Book
alder fill --decl Rooms.Show.view - <<'EOF'
<Layout.Page title={model.room.name}>
  <h1 id="name">{model.room.name}</h1>
  <ul id="bookings">
    {for b in model.bookings ->
      <li><span class="from">{Date.toText b.from}</span> <span class="to">{Date.toText b.to}</span></li>}
  </ul>
  <form submit={Book}>
    <Errors form />
    <Input field="from" label="From" />
    <Input field="to" label="To" />
    <button>Book</button>
  </form>
</Layout.Page>
EOF
alder fill --decl Rooms.Show.update - <<'EOF'
match msg {
  Book p -> {
    let existing <- Db.from Booking |> Db.where (Booking.room.eq model.room.id) |> Db.all
    if not (p.form.to > p.form.from) then Task.succeed (Reject "The end date must be after the start date.")
    else if Booking.conflicts p.form.from p.form.to existing then Task.succeed (Reject "The room is already booked in that period.")
    else {
      let _ <- Db.insert Booking { room: model.room.id, owner: session.user, from: p.form.from, to: p.form.to }
      Task.succeed (Redirect (Rooms.Show { id: model.room.id }))
    }
  },
}
EOF

# Bookings.Show
alder new page Bookings.Show '/bookings/{id}' --blank --requires user --param id:Booking.BookingId
alder fill --decl Bookings.Show.Model - <<'EOF'
{ booking: Booking, room: Text, today: Date }
EOF
alder fill Bookings.Show.load - <<'EOF'
{
  let found <- Db.get Booking params.id
  match found {
    Nothing -> Query.notFound,
    Just b -> if b.owner != session.user then Query.notFound else {
      let room <- Db.get Room b.room
      let today <- Date.today
      Query.succeed { booking: b, room: Maybe.withDefault "" (Maybe.map (\r -> "{r.name}") room), today: today }
    },
  }
}
EOF
alder fill Bookings.Show.view - <<'EOF'
<Layout.Page title="Booking">
  <h1>Booking</h1>
</Layout.Page>
EOF
alder new msg Bookings.Show Cancel
alder fill --decl Bookings.Show.view - <<'EOF'
<Layout.Page title="Booking">
  <h1>Booking</h1>
  <p id="room">{model.room}</p>
  <p id="from">{Date.toText model.booking.from}</p>
  <p id="to">{Date.toText model.booking.to}</p>
  {if model.today < model.booking.from then <button id="cancel" click={Cancel}>Cancel</button> else Html.none}
  <a href={Bookings.Index {}}>Back</a>
</Layout.Page>
EOF
alder fill --decl Bookings.Show.update - <<'EOF'
match msg {
  Cancel p -> if model.today < model.booking.from then {
    let _ <- Db.delete Booking model.booking.id
    Task.succeed (Redirect (Bookings.Index {}))
  } else Task.succeed Rerender,
}
EOF

# tests
alder new test Booking "next free date skips back-to-back bookings"
alder fill 'Booking."next free date skips back-to-back bookings"' - <<'EOF'
{
  let today <- Date.today
  let free = nextFree today [
    { id: BookingId "a", room: Room.RoomId "r", owner: User.UserId "u", from: today, to: Date.addDays 2 today },
    { id: BookingId "b", room: Room.RoomId "r", owner: User.UserId "u", from: Date.addDays 2 today, to: Date.addDays 5 today },
  ]
  Task.succeed (free == Date.addDays 5 today)
}
EOF
alder new test Rooms.Show "overlap is refused and back-to-back is stored"
alder fill 'Rooms.Show."overlap is refused and back-to-back is stored"' - <<'EOF'
{
  let me <- Db.insert User { email: Text.Email "me@example.com", name: Text.NonEmpty "Me", passwordHash: "" }
  let room <- Db.insert Room { name: Text.NonEmpty "Blue" }
  let today <- Date.today
  let d1 = Date.addDays 10 today
  let d3 = Date.addDays 12 today
  let d5 = Date.addDays 14 today
  let _ <- Db.insert Booking { room: room, owner: me, from: d1, to: d3 }
  let _ <- Test.login { user: Just me }
  let _ <- Test.get (Rooms.Show { id: room })
  let bad <- Test.submit NewBooking { from: Date.addDays 1 d1, to: d5 }
  let n1 <- Db.from Booking |> Db.count
  let _ <- Test.get (Rooms.Show { id: room })
  let ok <- Test.submit NewBooking { from: d3, to: d5 }
  let n2 <- Db.from Booking |> Db.count
  Task.succeed ({ bad: bad.answer, n1: n1, ok: ok.answer, n2: n2 } == { bad: FormRejected [{ field: "", message: "The room is already booked in that period." }], n1: 1, ok: Redirected (Rooms.Show { id: room }), n2: 2 })
}
EOF
